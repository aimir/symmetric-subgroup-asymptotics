\section{Same-source lifts and original-weight fusion}\label{sec:fusion}
\subsection{The binary permutation rank bound}
The binary rank input needed below has a direct degree-induction proof.
Write $d_2(P)=\dim_{\F_2}\Hom(P,C_2)$; for a finite $2$-group this
is also the dimension of its Frattini quotient.
\begin{lemma}\label{lem:binary-permutation-rank}
If a finite $2$-group $P$ acts faithfully on $b$ points, then
$d_2(P)\leq\lfloor b/2\rfloor$.
\end{lemma}
\begin{proof}
Induct on $b$, with degrees zero and one immediate. For an intransitive
action, restrict to a nonempty proper orbit of size $a$. Its actual image
acts faithfully on that orbit, and the restriction kernel acts faithfully
on the original complement. Character restriction and induction give
\[
 d_2(P)\leq d_2(\text{image})+d_2(\text{kernel})
 \leq\lfloor a/2\rfloor+\lfloor(b-a)/2\rfloor
 \leq\lfloor b/2\rfloor.
\]
For a transitive action, a cover of the original point stabilizer has
relative index two. Its actual blocks are pairs, so $b=2m$. Let $Q$
be the faithful action on the $m$ pairs and $K$ its original kernel.
The flip coordinates identify $K$ with a $P$-invariant subspace of
$\F_2^m$, retaining all correlations among the flips. Restriction gives
\[
 d_2(P)\leq d_2(Q)+\dim\Hom(K,\F_2)^P.
\]
If $m=1$, the top is trivial and $\dim K\leq1$. Otherwise the
transitive $2$-group $Q$ has a fixed-point-free central involution:
any nonidentity central element of order two is semiregular in a
faithful transitive action. Choose a lift $g\in P$. Every invariant
form on $K$ annihilates $(g-1)K$, hence rank--nullity gives
\[
 \dim\Hom(K,\F_2)^P
 \leq\dim K/(g-1)K
 =\dim\ker((g-1)|_K)
 \leq\dim\ker((g-1)|_{\F_2^m})=m/2.
\]
Induction bounds $d_2(Q)$ by $\lfloor m/2\rfloor$, so the total
is at most $m=b/2$. No splitting of the original extension is used.
\end{proof}

\subsection{The fractional Schur cost}
The following lemma applies to nonsplit group extensions and to
nonsemisimple modules.
\begin{lemma}[Conditional cocycle bound]\label{lem:cocycle}
Let $1\to A\to Q\to B\to1$, with $A$ elementary abelian of exponent
$p$, and let
$r(A)=\max_Xm_X(\Soc_B A)/\dim_{\End_BX}X$.
For $J\leq S_b$ and a fixed epimorphism $\beta:J\to B$, the number
of epimorphisms $J\to Q$ above $\beta$ is at most
\begin{equation}\label{eq:cocycle}
 |A|\,|H^1(B,A)|\,p^{(b/p)r(A)}.
\end{equation}
\end{lemma}
\begin{proof}
If a lift exists, choose one actual lift as origin. All homomorphic
lifts are a torsor for $Z^1(J,A_\beta)$; onto lifts are a subset.
Let $F=\ker\beta$. Inflation--restriction and the bound on
coboundaries give
\[
 |Z^1(J,A_\beta)|\leq|A|\,|H^1(B,A)|\,|\Hom(F,A)^J|.
\]
Choose a Sylow $p$-subgroup $P\leq F$ and put $D=N_J(P)$.
The Frattini argument gives $J=FD$, so $D\to B$ is onto.
Restriction embeds
\[
 \Hom(F,A)^J\lhook\joinrel\longrightarrow
 \Hom_D(P/\Phi(P),A).
\]
Indeed $P$ maps onto every $p$-group image of $F$.
The source here has the actual $D$-action; an action of $B$ on it
is not assumed. Filter $W=P/\Phi(P)$ by a composition series.
Left exactness of $\Hom(-,A)$ gives
\[
 \dim\Hom_D(W,A)\leq
 \sum_X[W:X]m_X\dim_{\F_p}\End_DX\leq r(A)\dim W.
\]
Types not inflated from $B$ contribute zero. Since $D\to B$ is onto,
the remaining simple dimensions and endomorphism fields are precisely
those used in $r(A)$. Finally $\dim W\leq b/p$ by the permutation
$p$-group abelianization bound, whose binary case is
Lemma~\ref{lem:binary-permutation-rank}. This proves \eqref{eq:cocycle}.
\end{proof}

\begin{proposition}[Central cut and graph moment]\label{prop:prefix}
Let $1\to A\to Q\to B\to1$ have elementary binary kernel and let
$C\leq A^B$, $c=\dim C$. Suppose $T\leq S_s$ is a specified
permutation cover $\pi:T\to B$. Put
\[
 D_C=|A/C|\,|H^1(B,A/C)|,\quad
 \lambda=\tfrac12r(A/C),\quad v=s+2c,\quad
 \Phi(J)=|\Epi(J,B)|2^{c d_2(J)}.
\]
For $A/C=0$ take $D_C=1$. Then, for every $b$ and integer $q\geq1$,
\begin{equation}\label{eq:prefix}
 |\Epi(J,Q)|\leq D_C2^{\lambda b}\Phi(J),\qquad
 \sum_{J\leq S_b}\Phi(J)^q\leq s_{b+qv}.
\end{equation}
\end{proposition}
\begin{proof}
Since $C\leq Z(Q)$, each actual map $J\to Q/C$ has either no lift
or a torsor of homomorphic lifts under $\Hom(J,C)$. Apply
Lemma~\ref{lem:cocycle} to the actual extension
$1\to A/C\to Q/C\to B\to1$ and sum over the same maps to $B$.

A term $\Phi(J)^q$ selects $q$ pairs of literal maps
$(\beta_i:J\twoheadrightarrow B,\chi_i:J\to C_2^c)$.
Encode them by the subgroup
\[
 \{(t_1,e_1,\ldots,t_q,e_q,j):
    \pi(t_i)=\beta_i(j),\ e_i=\chi_i(j)\text{ for all }i\}
       \leq(T\times C_2^c)^q\times J.
\]
Realize each $C_2^c$ on $c$ disjoint pairs. Projection onto the fixed
coordinate sets recovers $J$ and every selected map. Thus the encoding
is injective into the subgroups of $S_{b+qv}$. Repeated maps, dependent
characters, and proper joint subdirect images are allowed.
\end{proof}

\begin{proposition}[A target-order alternative]\label{prop:target-order-fusion}
Let $Q$ be a finite $2$-group of order $2^a$. For every finite group
$J$ and all integers $b,q\geq0$,
\[
 |\Hom(J,Q)|\leq2^{a d_2(J)},\qquad
 \sum_{J\leq S_b}\bigl(2^{a d_2(J)}\bigr)^q\leq s_{b+2aq}.
\]
Consequently this supplies the fusion parameters
$D=1$, $\lambda=0$, $v=2a$ whenever the original physical width
satisfies $2a<w$.
\end{proposition}
\begin{proof}
Every nontrivial finite $2$-group has a central subgroup $C$ of
order two. Each nonempty fibre of
$\Hom(J,Q)\to\Hom(J,Q/C)$ is a torsor under $\Hom(J,C)$:
the pointwise difference of two lifts is a homomorphism because
$C$ is central. Its cardinality is $2^{d_2(J)}$.
Induction on $a$ proves the first inequality, including empty fibres
and nonsplit extensions. For the second, encode the $aq$ binary
characters together as a graph over the same literal $J$, realizing
each binary coordinate on a pair of new points.
\end{proof}
This argument counts actual maps; it takes no quotient by target
automorphisms and does not require a faithful action of $Q$ on $2a$
points. Its width condition is distinct from the character criterion
used later.

\subsection{A complete finite-menu estimate}
\begin{theorem}[Original-weight fusion]\label{thm:fusion}
Fix a finite collection of pairs $(U,N)$, where $U\leq S_w$ is
transitive, $w$ is even, and $N\lhd U$. For each pair suppose there
are fixed $D,v,\lambda$ and a nonnegative function $\Phi$ such that
\[
 |\Epi(J,U/N)|\leq D2^{\lambda b}\Phi(J),\qquad
 \sum_{J\leq S_b}\Phi(J)^q\leq s_{b+qv}\quad(q\geq1),
 \qquad g=w-v-8\lambda>0.
\]
Let $\mathcal U_n$ be any family of actual subgroups of $S_n$ each
having an orbit of type $U$ whose intersection axis is one of the
specified $N$. Then
\begin{equation}\label{eq:finite-fusion}
 \frac{|\mathcal U_n|}{L_n}
 \leq\eta(n)+\sum_{b<n}K(n,b)\frac{s_b}{L_b},
 \qquad\eta(n)=O(2^{-a n^2}),\quad
 \sum_bK(n,b)=O(2^{-a' n})
\end{equation}
for some $a,a'>0$. The weights are those of the original actions.
The result is uniform in ambient parity, but its constants may depend
on the fixed finite menu.
\end{theorem}
\begin{proof}
For each $U$, first close the selected literal axes under
$N_{S_w}(U)$. This finite enlargement leaves the existential physical
orbit event unchanged. Choose one certificate on each resulting
normalizer orbit and transport it through the induced quotient
isomorphisms; thus $D,v,\lambda,\Phi$ can be constant on that orbit.
All sums below refer to this closed literal-axis menu.

On a pointed orbit the subgroup has exact Goursat form
\[
 \{(u,j)\in U\times J:uN=\beta(j)\}
\]
for a unique literal epimorphism $\beta:J\to U/N$, once the quotient
map is fixed. This counts actual maps, so there is no additional
division by $|\Aut(U/N)|$. First apply the pointing factor
$n!/(b!a(U))$ to the total normalizer-stable sum over literal axes and
complete sources. A single axis need not be fixed by the action
normalizer. Only after this division do we split the numerical sum
entry by entry; the envelopes need not be conjugacy invariant.
For an entry put
$e=g/16$, $a_0=v/8+e$, and call its complete source hot when
$\Phi(J)>2^{a_0b}$.
If $v=0$, the moment bound has the same finite right-hand side $s_b$
for every $q$. Hence $\Phi(J)\leq1$ for every $J$, and the strict hot
set is empty. We may therefore assume $v>0$ in the hot estimate below.

For a hot source and any integer $q\geq1$,
\[
 \sum_{J\ {m hot}}|\Epi(J,U/N)|
 \leq D2^{\lambda b-(q-1)a_0b}s_{b+qv}.
\]
Choose $q=\max(1,\lceil8eb/v^2\rceil)$. The exact identity
\begin{align}\label{eq:hot-square}
 &\frac{(b+qv)^2}{16}-(q-1)a_0b+\lambda b
              -\frac{(b+w)^2}{16}\notag\\
 &\hspace{8mm}=-\frac{4e^2b^2}{v^2}-eb-\frac{w^2}{16}
       +\frac{v^2}{16}\left(q-\frac{8eb}{v^2}\right)^2
\end{align}
has a strictly negative quadratic term. The all-pairs monomial gives
$c_R\geq1/(2^RR!)$. Thus, writing $R=\lfloor n/2\rfloor$ and
using $b\geq w$ so that $R\leq b$,
\[
 \frac{n!}{b!L_n}
 \leq\phi^{-1}2^{-R^2/4+R+1/4}
 \leq\phi^{-1}2^{-n^2/16+5n/8+1/4}.
\]
Together with \eqref{eq:coarse}, this contributes only
$O(n^{3/2}+n)$ to the normalized logarithm. No saddle estimate is
needed for this normalization.
As all entry parameters are fixed, the complete finite hot union is
$O(2^{-a n^2})$.

On its complement,
$|\Epi(J,U/N)|\leq D2^{(w/8-g/16)b}$. The cold kernel is bounded by
the sum over entries of
\begin{equation}\label{eq:cold-kernel}
 \Delta(n,b)\frac{D}{a(U)}2^{(w/8-g/16)b},\qquad b=n-w.
\end{equation}
For $w=2h$, \eqref{eq:coefficient-recurrence} and the Gaussian
bounds give the explicit estimate
\[
 \Delta(n,b)\leq
 \phi^{-2}2^{(h+1)/4}(n+1)^{h+1}2^{-hb/4-h^2/4}.
\]
The guarded odd coefficient obeys the same backward-shift bound,
including rank zero, so the estimate holds in both parities. Each summand
in \eqref{eq:cold-kernel} is therefore
$2^{-gb/16+O_w(\log(n+2))}$. The finite sum proves the row estimate.

The hot and cold contributions partition the local numerical sum
after pointing. Summing over the finite action menu may overcount,
but only by the positive sum already bounded. Restriction by any
earlier disjoint family can only reduce it. The actual complement
$J$ includes every later orbit and every terminal attachment, as
required by \eqref{eq:pointing}.
\end{proof}

\begin{proposition}[Direct first-moment fusion]\label{prop:direct-first-moment-fusion}
Fix a finite original action/normal-axis menu as above. For each entry
assume the local envelope and only the first moment
\[
 |\Epi(J,U/N)|\leq D2^{\lambda b}\Phi(J),\qquad
 \sum_{J\leq S_b}\Phi(J)\leq s_{b+v}.
\]
Suppose $w=2h$, $v=2r<w$, and $g=w-v-8\lambda>0$.
For the same actual orbit family, with any retained survival predicates
invariant under the relabellings in the pointing formula, there is a
nonnegative forward kernel satisfying
\[
 \frac{|\mathcal U_n|}{L_n}
 \leq\sum_{m<n}K_{\rm dir}(n,m)\frac{s_m}{L_m},\qquad
 \sum_{m<n}K_{\rm dir}(n,m)=O(2^{-\kappa n})
\]
for some $\kappa>0$. This conclusion does not use the coarse estimate
\eqref{eq:coarse}.
\end{proposition}
\begin{proof}
Apply the original pointing factor to the complete literal-axis sum,
as in the preceding proof, and then use the first moment. An entry
continues at $m=b+v=n-w+v<n$, with coefficient
\[
 \Gamma_{U,N}(b)=
 \frac{(b+w)!}{b!\,a(U)}D2^{\lambda b}
 \frac{L_{b+v}}{L_{b+w}}.
\]
Define $K_{\rm dir}(n,m)$ by summing these coefficients over all
literal entries with this target degree, taking an entry to be zero
when $n<w$. Distinct normals and actions
remain separate summands, even when their target degrees agree.
The divisor is still $a(U)$, and no additional quotient by target
automorphisms or marker choices occurs.

Put $k=h-r>0$. Since $n=m+2k$, exact factorial cancellation gives
\[
 \Gamma_{U,N}(b)
 =\frac{D}{a(U)}\frac{m!}{b!}2^{\lambda b}\Delta(n,m).
\]
The fixed-width benchmark estimate above applies in either parity,
and $m!/b!\leq m^v$. Retaining its quadratic width term gives the
uniform estimate, valid for every $b\geq0$,
\begin{equation}\label{eq:direct-uniform-coefficient}
 \Gamma_{U,N}(b)
 \leq \phi^{-2}\frac{D}{a(U)}(n+1)^{(w+v)/2+1}
 2^{-gb/8-(w^2-v^2)/16+(w-v+2)/8}.
\end{equation}
Indeed the exponent before simplification is
$\lambda b-km/4-k^2/4+(k+1)/4$.
Consequently, for an entry-dependent constant,
\[
 \Gamma_{U,N}(b)
 \leq C(n+1)^{v+k+1}2^{\lambda b-km/4}
 \leq C(n+1)^{v+k+1}2^{-gb/8}.
\]
Absorb the fixed polynomial into the exponential and sum the fixed
finite menu. This proves the row estimate without estimating any
unknown subgroup count $s_m$.
\end{proof}
The order markers $v=2\log_2|U/N|$ and the character marker $v=0$
fit this even-prefix argument whenever their strict width conditions
hold. Any class-count input used to establish the character envelope
remains necessary. The proposition supplies neither a physical cover
by the menu nor uniform control when the widths, menu, or lift constants
grow with $n$; those wider regimes remain separate obligations.

\begin{corollary}[Nonbinary pair actions]\label{cor:pair-fusion}
Let $U$ be a transitive action on $2s$ points with an invariant pair
system whose faithful top $T\leq S_s$ is nonbinary, with even $s\geq24$.
For every $N\lhd U$, write $K$ for the pair kernel and set
\[
 A=KN/N\cong K/(K\cap N),\qquad B=U/(KN).
\]
This entry has the certificate of Theorem~\ref{thm:fusion}, with
$g\geq s/48$. Hence every fixed finite menu of these entries admits
\eqref{eq:finite-fusion}.
\end{corollary}
\begin{proof}
The pair kernel is a submodule of $\F_2^s$; the action on its indicated
section factors through $B$. Apply Theorem~\ref{thm:capacity-intro}
and Proposition~\ref{prop:prefix}. With $c=\dim C$,
\[
 g=2s-(s+2c)-4r(A/C)\geq s/48.
\]
There was no assumption that $N\leq K$, nor that the common quotient
or its subdirect cores are abelian. For labelled transport, fix one
witness on each original-normalizer orbit of entries and fixed
transporters before counting $J$. A normalizer-equivariant choice of
an individual witness is not needed, and the witnesses are not counted
as additional decorations.
\end{proof}

\begin{remark}[Scope of the positive enlargement]
The original quotient, extension class, and restricted group
transgression determine which lifts exist. The cocycle torsor bounds
are applied after those data have been fixed. The capacity proof uses
the distinct module annihilator in \eqref{eq:fixed-quotient}.
The theorem does not grant bounds for arbitrary extra weights on the
source groups, and a growing menu requires an additional uniform
estimate for its size and the constants $D$. Only $\Phi$, not the
cocycle tail $2^{\lambda b}$, is raised to the moment power.
\end{remark}
